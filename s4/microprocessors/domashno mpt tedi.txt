ExitProcess proto:DWORD
.data
A               dd      23,34,37,38,41,44,57,61,64,72   
BR              dd      0   
N               dd      10
RANGE_START     equ     30
RANGE_END       equ     55
.code
start:  
    lea esi, A 
    mov ecx, N
    mov ebx, BR
    call counting
    add BR, eax
    Invoke ExitProcess, 0

counting proc
    ProcessArray:
        mov eax, [esi]
        add esi, 4
        cmp eax, RANGE_START
        jl notInRange
        cmp eax, RANGE_END
        jg notInRange
        inc ebx
    notInRange:
        loop ProcessArray
    mov BR, ebx
    ret
counting endp

end start

.model flat, stdcall
includelib \masm32\lib\kernel32.lib
ExitProcess proto:DWORD

.data
NameSurname db 'teodora stoycheva', 0 
countA dd 0 

.code
start:
    cld 
    lea esi, NameSurname 
    mov ecx, 0 
    mov eax, 0 

count_loop:
    lodsb 
    cmp al, 0 
    je end_count_loop
    cmp al, 'a' 
    jne not_a
    inc ecx 
not_a:
    jmp count_loop

end_count_loop:
    mov [countA], ecx
    Invoke ExitProcess, 0
end start

    cmp r3, #'a'

not_a:
