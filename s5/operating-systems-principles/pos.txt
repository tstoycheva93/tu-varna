poxis threads - pthreads
kompilirane: gcc -o <exe> <source> -lpthread
gcc -o zad zad.c -lpthread
biblioteka: pthread
header: pthread.h

suzdavane na nishka:
int pthread_create - zapazva unikalno id

zavrshvane:
exit(), 
void pthread_exit(void *status)

#include<pthread.h>
#include<stdio.h>

void *Plus() {
int k,i;
for (int i=0; i<=100; i++) {
printf("+\n");
for (k=0;k<100;k++);
}
}

void *Minus() {
int k,i;
for (int i=0; i<=100; i++) {
printf("-\n");
for (k=0;k<100;k++);
}
}

void *Dot() {
int k,i;
for (int i=0; i<=100; i++) {
printf(".\n");
for (k=0;k<100;k++);
}
}

int main() {
pthread_t t1, t2, t3;
pthread_create (&t1, NULL, Plus, NULL);
pthread_create (&t2, NULL, Minus, NULL);
pthread_create (&t3, NULL, Dot, NULL);

printf("\n");

printf("Creating thread %lu\n", t1);
printf("Creating thread %lu\n", t2);
printf("Creating thread %lu\n", t3);

pthread_exit(NULL);
}
