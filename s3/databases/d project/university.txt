CREATE TABLE Student 
(
    student_id CHAR(8) NOT NULL,
    sutdent_name VARCHAR2(20) NOT NULL,
    course INTEGER NOT NULL,
    student_email VARCHAR2(100),
    major_id VARCHAR2(10) NOT NULL,
    PRIMARY KEY(student_id)
);

CREATE TABLE Major 
(
    major_id VARCHAR2(10) NOT NULL, 
    major_name VARCHAR2(10) NOT NULL, 
    PRIMARY KEY(major_id)
);

-- създаване на връзка между две таблица P.K. => F.K.
 
ALTER TABLE Student  -- име на таблица, в която е F.K.
ADD CONSTRAINT  Student_Major_FK  -- име на правилото 
FOREIGN KEY (major_id) -- име на колона, която е Ф.К за таблицата
REFERENCES major(major_id); -- име на таблица, където е нашия П.К. и съответно самаа колона П.К.

CREATE TABLE Student_Subject 
(
    student_id CHAR(8) NOT NULL, 
    subject_id VARCHAR2(5) NOT NULL
);

ALTER TABLE Student_Subject  -- име на таблица, в която е F.K.
ADD CONSTRAINT  Student_Subject_Sudent_FK  -- име на правилото 
FOREIGN KEY (student_id) -- име на колона, която е Ф.К за таблицата
REFERENCES Student(student_id);


CREATE TABLE Subject
(
    subject_id VARCHAR2(10) NOT NULL, 
    semester INTEGER NOT NULL,
    subject_name VARCHAR2(10),
    PRIMARY KEY (subject_id)
);

ALTER TABLE Student_Subject  -- име на таблица, в която е F.K.

FOREIGN KEY (subject_id) -- име на колона, която е Ф.К за таблицата

CREATE TABLE Teacher_Subject 
(

    subject_id VARCHAR2(10CHAR) NOT NULL
